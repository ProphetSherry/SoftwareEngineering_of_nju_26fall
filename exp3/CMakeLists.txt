cmake_minimum_required(VERSION 3.16)
project(CampusTickets VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BUILD_GUI "Build the Qt desktop application" ON)
option(ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
find_package(SQLite3 REQUIRED)
add_library(campus_core src/database.cpp src/service.cpp)
target_include_directories(campus_core PUBLIC include)
target_link_libraries(campus_core PUBLIC SQLite::SQLite3 crypt)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
target_compile_options(campus_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(ENABLE_SANITIZERS)
  target_compile_options(campus_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(campus_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(campus_cli src/cli.cpp)
target_link_libraries(campus_cli PRIVATE campus_core)
if(BUILD_GUI)
  find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets)
  find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets)
  add_executable(campus_gui src/main.cpp src/window.cpp)
  target_link_libraries(campus_gui PRIVATE campus_core Qt${QT_VERSION_MAJOR}::Widgets)
endif()
include(CTest)
find_package(Threads REQUIRED)
if(BUILD_TESTING)
  add_executable(campus_tests tests/core_tests.cpp)
  target_link_libraries(campus_tests PRIVATE campus_core Threads::Threads)
  add_test(NAME core_workflows COMMAND campus_tests)
endif()
add_executable(campus_benchmark tools/benchmark.cpp)
target_link_libraries(campus_benchmark PRIVATE campus_core Threads::Threads)
if(BUILD_GUI AND BUILD_TESTING)
  add_executable(campus_gui_tests tests/gui_tests.cpp src/window.cpp)
  target_link_libraries(campus_gui_tests PRIVATE campus_core Qt${QT_VERSION_MAJOR}::Widgets)
  add_test(NAME organizer_chinese_input COMMAND campus_gui_tests)
  set_tests_properties(organizer_chinese_input PROPERTIES
    ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 20)
endif()
if(BUILD_GUI AND BUILD_TESTING)
  add_executable(campus_button_tests tests/button_tests.cpp src/window.cpp)
  target_link_libraries(campus_button_tests PRIVATE campus_core Qt${QT_VERSION_MAJOR}::Widgets)
  add_test(NAME all_ui_buttons COMMAND campus_button_tests
    "${CMAKE_CURRENT_SOURCE_DIR}/evidence/button-tests.json")
  set_tests_properties(all_ui_buttons PROPERTIES
    ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 180)
endif()
